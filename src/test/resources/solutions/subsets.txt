        List<List<Integer>> result = new ArrayList<>();
        result.add(new ArrayList<>());

        for (int num : nums) {
            int size = result.size();
            for (int i = 0; i < size; i++) {
                List<Integer> withNum = new ArrayList<>(result.get(i));
                withNum.add(num);
                result.add(withNum);
            }
        }

        return result;
